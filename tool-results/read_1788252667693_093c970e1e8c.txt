    89→  void _toggleEdit() {

   102→        _priceController.clear();

   148→  void _onProductSelected(Product product) {
   149→    setState(() {
   150→      final isSameProduct = _selectedProduct?.id == product.id;
   151→      _selectedProduct = product;
   152→      if (!isSameProduct) {
   153→        // Pré-remplit le prix avec le dernier prix vendu à ce client
   154→        final lastPrice = _clientPrices[product.name.trim().toLowerCase()];
   155→        if (lastPrice != null) {
   156→          _priceController.text = _priceInputValue(lastPrice);
   157→        } else {
   158→          _priceController.clear();
   159→        }
   160→      }
   161→    });
   162→  }
   163→
   164→  String _priceInputValue(double value) =>
   165→      value % 1 == 0 ? value.toInt().toString() : value.toString();

   167→  void _addNewItem() {
   168→    final quantity = int.tryParse(_quantityController.text.trim());
   169→    if (quantity == null || quantity <= 0) {
   170→      ScaffoldMessenger.of(context).showSnackBar(
   171→        const SnackBar(
   172→          content: Text('Veuillez entrer une quantité valide'),
   173→          backgroundColor: Colors.red,
   174→          behavior: SnackBarBehavior.floating,
   175→        ),
   176→      );
   177→      return;
   178→    }
   179→
   180→    double? price = double.tryParse(_priceController.text.trim());
   181→    if (price == null || price <= 0) {
   182→      price = 1.0;
   183→      _priceController.text = '1';
   184→    }
   185→
   186→    if (_selectedProduct == null) {
   187→      ScaffoldMessenger.of(context).showSnackBar(
   188→        const SnackBar(
   189→          content: Text('Veuillez sélectionner un produit'),
   190→          backgroundColor: Colors.orange,
   191→          behavior: SnackBarBehavior.floating,
   192→        ),
   193→      );
   194→      return;
   195→    }
   196→
   197→    final newItems = List<InvoiceItem>.from(_invoice.items)
   198→      ..add(InvoiceItem(
   199→        name: _selectedProduct!.name,
   200→        quantity: quantity,
   201→        unitPrice: price,
   202→      ));
   203→    setState(() {
   204→      _invoice = _invoice.copyWith(items: newItems);
   205→      _isAddingItem = false;
   206→      _selectedProduct = null;
   207→      _quantityController.clear();
   208→      _priceController.clear();
   209→    });
   210→  }

   325→  Future<void> _saveChanges() async {
   326→    setState(() => _isSaving = true);
   327→    // Construit explicitement la facture (plutôt que copyWith) pour permettre
   328→    // d'effacer réellement les notes quand le champ est vidé.
   329→    double parsePositive(String text) {
   330→      final value = double.tryParse(text.trim().replaceAll(',', '.')) ?? 0;
   331→      return value < 0 ? 0 : value;
   332→    }
   333→
   334→    final updatedInvoice = Invoice(
   335→      id: _invoice.id,
   336→      clientName: _clientNameController.text.trim().isEmpty
   337→          ? 'Client inconnu'
   338→          : _clientNameController.text.trim(),
   339→      createdAt: _invoice.createdAt,
   340→      items: _invoice.items.toList(),
   341→      status: _invoice.status,
   342→      notes: _notesController.text.trim().isEmpty ? null : _notesController.text.trim(),
   343→      discount: parsePositive(_discountController.text),
   344→      taxRate: parsePositive(_taxRateController.text).clamp(0, 100).toDouble(),
   345→    );
   346→
   347→    final all = await _storageService.loadInvoices();
   348→    final idx = all.indexWhere((inv) => inv.id == updatedInvoice.id);
   349→    if (idx != -1) {
   350→      all[idx] = updatedInvoice;
   351→      await _storageService.saveInvoices(all);
   352→    }
   353→
   354→    setState(() {
   355→      _invoice = updatedInvoice;
   356→      _isEditing = false;
   357→      _isSaving = false;
   358→    });
   359→
   360→    widget.onInvoiceUpdated();
   361→    if (!mounted) return;
   362→    ScaffoldMessenger.of(context).showSnackBar(
   363→      const SnackBar(content: Text('Facture mise à jour'), backgroundColor: Colors.green, behavior: SnackBarBehavior.floating),
   364→    );
   365→  }

   367→  Future<void> _deleteInvoice() async {
   368→    final confirm = await showDialog<bool>(
   369→      context: context,
   370→      builder: (context) => AlertDialog(
   371→        shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(Constants.cardRadius)),
   372→        title: const Text('Supprimer la facture ?'),
   373→        content: const Text('Cette action est irréversible.'),
   374→        actions: [
   375→          TextButton(onPressed: () => Navigator.pop(context, false), child: const Text('Annuler')),
   376→          TextButton(onPressed: () => Navigator.pop(context, true), child: const Text('Supprimer', style: TextStyle(color: Colors.red))),
   377→        ],
   378→      ),
   379→    );
   380→    if (confirm != true) return;
   381→
   382→    final all = await _storageService.loadInvoices();
   383→    all.removeWhere((inv) => inv.id == _invoice.id);
   384→    await _storageService.saveInvoices(all);
   385→
   386→    if (!mounted) return;
   387→    widget.onInvoiceUpdated();
   388→    Navigator.pop(context);
   389→  }

   809→class _EditableItemTile extends StatefulWidget {
   810→  final int index;
   811→  final InvoiceItem item;
   812→  final Function(InvoiceItem) onUpdate;
   813→  final VoidCallback onDelete;
   814→  const _EditableItemTile({required this.index, required this.item, required this.onUpdate, required this.onDelete});
   815→
   816→  @override
   817→  State<_EditableItemTile> createState() => _EditableItemTileState();
   818→}
   819→
   820→class _EditableItemTileState extends State<_EditableItemTile> {
   821→  late TextEditingController _nameController;
   822→  late TextEditingController _quantityController;
   823→  late TextEditingController _priceController;
   824→
   825→  @override
   826→  void initState() {
   827→    super.initState();
   828→    _nameController = TextEditingController(text: widget.item.name);
   829→    _quantityController = TextEditingController(text: widget.item.quantity.toString());
   830→    _priceController = TextEditingController(text: widget.item.unitPrice.toString());
   831→  }
   832→
   833→  @override
   834→  void didUpdateWidget(covariant _EditableItemTile oldWidget) {
   835→    super.didUpdateWidget(oldWidget);
   836→    // Re-synchronise les champs quand l'article change (ex: suppression multiple
   837→    // qui décale les index) pour éviter que les valeurs se mélangent.
   838→    if (!identical(oldWidget.item, widget.item)) {
   839→      _nameController.text = widget.item.name;
   840→      _quantityController.text = widget.item.quantity.toString();
   841→      _priceController.text = widget.item.unitPrice.toString();
   842→    }
   843→  }
   844→
   845→  @override
   846→  void dispose() {
   847→    _nameController.dispose();
   848→    _quantityController.dispose();
   849→    _priceController.dispose();
   850→    super.dispose();
   851→  }
   852→
   853→  void _update() {
   854→    final quantity = int.tryParse(_quantityController.text);
   855→    final price = double.tryParse(_priceController.text);
   856→    if (quantity == null || quantity <= 0 || price == null || price <= 0) {
   857→      ScaffoldMessenger.of(context).showSnackBar(const SnackBar(content: Text('Valeurs invalides'), backgroundColor: Colors.red));
   858→      return;
   859→    }
   860→    widget.onUpdate(widget.item.copyWith(
   861→      name: _nameController.text.trim().isEmpty ? 'Sans nom' : _nameController.text.trim(),
   862→      quantity: quantity,
   863→      unitPrice: price,
   864→    ));
   865→  }
   866→
   867→  @override
   868→  Widget build(BuildContext context) {
   869→    return Column(
   870→      crossAxisAlignment: CrossAxisAlignment.start,
   871→      children: [
   872→        Row(
   873→          children: [
   874→            Container(width: 28, height: 28, decoration: const BoxDecoration(color: Color(0xFFDBEAFE), shape: BoxShape.circle), child: Center(child: Text('${widget.index + 1}', style: const TextStyle(fontWeight: FontWeight.bold, color: Color(0xFF2563EB), fontSize: 12)))),
   875→            const SizedBox(width: 10),
   876→            Expanded(child: TextField(controller: _nameController, decoration: const InputDecoration(labelText: 'Nom', border: OutlineInputBorder(), isDense: true))),
   877→          ],
   878→        ),
   879→        const SizedBox(height: 10),
   880→        Row(
   881→          children: [
   882→            Expanded(child: TextField(controller: _quantityController, keyboardType: TextInputType.number, inputFormatters: [FilteringTextInputFormatter.digitsOnly], decoration: const InputDecoration(labelText: 'Quantité', border: OutlineInputBorder(), isDense: true))),
   883→            const SizedBox(width: 8),
   884→            Expanded(child: TextField(controller: _priceController, keyboardType: const TextInputType.numberWithOptions(decimal: true), inputFormatters: [FilteringTextInputFormatter.allow(RegExp(r'^\d*\.?\d*'))], decoration: const InputDecoration(labelText: 'Prix unitaire', border: OutlineInputBorder(), isDense: true))),
   885→            IconButton(icon: const Icon(Icons.check, color: Colors.green), onPressed: _update, tooltip: 'Valider'),
   886→            IconButton(icon: const Icon(Icons.delete, color: Colors.red), onPressed: widget.onDelete, tooltip: 'Supprimer'),
   887→          ],
   888→        ),
   889→      ],
   890→    );
   891→  }
