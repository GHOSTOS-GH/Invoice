    16→  dateDesc,
    17→  dateAsc,
    18→  totalDesc,
    19→  totalAsc,
    20→  clientAsc,

    24→class HistoryScreen extends StatefulWidget {

    32→class HistoryScreenState extends State<HistoryScreen> with AutomaticKeepAliveClientMixin {
    33→  final _storageService = StorageService();
    34→  final _pdfService = PdfService();
    35→  List<Invoice> _invoices = [];
    36→  List<Invoice> _filteredInvoices = [];
    37→  bool _isLoading = true;
    38→  bool _loadingError = false;
    39→  final TextEditingController _searchController = TextEditingController();
    40→  String _searchQuery = '';
    41→  bool _isSelectionMode = false;
    42→  Set<String> _selectedIds = {};
    43→
    44→  // Filtres d'état
    45→  bool _filterEnCours = false;
    46→  bool _filterEnLivraison = false;
    47→  bool _filterLivree = false;
    48→  bool _filterArchivee = false;
    49→
    50→  // Filtre par période de dates (ex: factures du 10/02 au 11/02)
    51→  DateTimeRange? _filterDateRange;
    52→
    53→  // Option de tri
    54→  SortOption _currentSort = SortOption.dateDesc;

    56→  // Pagination

    60→  @override

    63→  @override
    64→  void initState() {
    65→    super.initState();
    66→    _loadInvoices();
    67→    _searchController.addListener(_onSearchChanged);
    68→    _scrollController.addListener(_onScroll);
    69→  }
    70→
    71→  @override
    72→  void dispose() {
    73→    _searchController.removeListener(_onSearchChanged);
    74→    _searchController.dispose();
    75→    _scrollController.dispose();
    76→    super.dispose();

    79→  void _onScroll() {
    80→    if (_scrollController.position.pixels >= _scrollController.position.maxScrollExtent - 200) {
    81→      if (_displayedCount < _filteredInvoices.length) {
    82→        setState(() {
    83→          _displayedCount = (_displayedCount + 50).clamp(0, _filteredInvoices.length);
    84→        });
    85→      }
    86→    }
    87→  }
    88→
    89→  void _onSearchChanged() {
    90→    setState(() {
    91→      _searchQuery = _searchController.text.toLowerCase();
    92→      _filterInvoices();
    93→      _displayedCount = 50;
    94→    });
    95→  }
    96→
    97→  void _filterInvoices() {
    98→    _filteredInvoices = List.from(_invoices);
    99→
   100→    // Filtres de statut
   101→    final hasActiveStatusFilter = _filterEnCours || _filterEnLivraison || _filterLivree || _filterArchivee;
   102→    if (hasActiveStatusFilter) {
   103→      _filteredInvoices = _filteredInvoices.where((invoice) {
   104→        if (_filterEnCours && invoice.status == InvoiceStatus.enCours) return true;
   105→        if (_filterEnLivraison && invoice.status == InvoiceStatus.enLivraison) return true;
   106→        if (_filterLivree && invoice.status == InvoiceStatus.livree) return true;
   107→        if (_filterArchivee && invoice.status == InvoiceStatus.archivee) return true;
   108→        return false;
   109→      }).toList();
   110→    }
   111→
   112→    // Filtre par période de dates
   113→    if (_filterDateRange != null) {
   114→      final start = DateTime(
   115→        _filterDateRange!.start.year,
   116→        _filterDateRange!.start.month,
   117→        _filterDateRange!.start.day,
   118→      );
   119→      final end = DateTime(
   120→        _filterDateRange!.end.year,
   121→        _filterDateRange!.end.month,
   122→        _filterDateRange!.end.day,
   123→        23, 59, 59,
   124→      );
   125→      _filteredInvoices = _filteredInvoices.where((invoice) {
   126→        final d = invoice.createdAt;
   127→        return !d.isBefore(start) && !d.isAfter(end);
   128→      }).toList();
   129→    }
   130→
   131→    // Recherche étendue (client, articles, notes, numéro)
   132→    if (_searchQuery.isNotEmpty) {
   133→      _filteredInvoices = _filteredInvoices.where((invoice) {
   134→        if (invoice.clientName.toLowerCase().contains(_searchQuery)) return true;
   135→        if (invoice.id.toLowerCase().contains(_searchQuery)) return true;
   136→        if (invoice.notes != null && invoice.notes!.toLowerCase().contains(_searchQuery)) return true;
   137→        for (var item in invoice.items) {
   138→          if (item.name.toLowerCase().contains(_searchQuery)) return true;
   139→        }
   140→        return false;
   141→      }).toList();
   142→    }
   143→
   144→    _sortInvoices();
   145→    setState(() {});
   146→  }
   147→
   148→  void _sortInvoices() {
   149→    switch (_currentSort) {
   150→      case SortOption.dateDesc:
   151→        _filteredInvoices.sort((a, b) => b.createdAt.compareTo(a.createdAt));
   152→        break;
   153→      case SortOption.dateAsc:
   154→        _filteredInvoices.sort((a, b) => a.createdAt.compareTo(b.createdAt));
   155→        break;
   156→      case SortOption.totalDesc:
   157→        _filteredInvoices.sort((a, b) => b.total.compareTo(a.total));
   158→        break;
   159→      case SortOption.totalAsc:
   160→        _filteredInvoices.sort((a, b) => a.total.compareTo(b.total));
   161→        break;
   162→      case SortOption.clientAsc:
   163→        _filteredInvoices.sort((a, b) => a.clientName.compareTo(b.clientName));
   164→        break;
   165→      case SortOption.statusAsc:
   166→        _filteredInvoices.sort((a, b) => a.status.index.compareTo(b.status.index));
   167→        break;
   168→    }
   169→  }
   170→
   171→  Future<void> _loadInvoices() async {
   172→    setState(() {
   173→      _isLoading = true;
   174→      _loadingError = false;
   175→    });
   176→    try {
   177→      final invoices = await _storageService.loadInvoices();
   178→      if (!mounted) return;
   179→      setState(() {
   180→        _invoices = invoices;
   181→        _filterInvoices();
   182→        _isLoading = false;
   183→      });
   184→    } catch (e) {
   185→      if (!mounted) return;
   186→      setState(() {
   187→        _isLoading = false;
   188→        _loadingError = true;
   189→      });
   190→    }
   191→  }
   192→
   193→  void refreshData() {
   194→    _loadInvoices();
   195→  }

   197→  String _formatDate(DateTime date) => DateFormat('dd/MM/yyyy à HH:mm', 'fr_FR').format(date);

   200→    Navigator.of(context).push(
   201→      PageRouteBuilder(
   202→        transitionDuration: const Duration(milliseconds: 300),
   203→        pageBuilder: (context, animation, secondaryAnimation) => InvoiceDetailScreen(

   205→          onInvoiceUpdated: widget.onInvoiceUpdated,
   206→        ),
   207→        transitionsBuilder: (context, animation, secondaryAnimation, child) {
   208→          return SlideTransition(
   209→            position: Tween<Offset>(
   210→              begin: const Offset(1.0, 0.0),
   211→              end: Offset.zero,
   212→            ).animate(CurvedAnimation(parent: animation, curve: Curves.easeInOut)),
   213→            child: child,
   214→          );
   215→        },
   216→      ),
   217→    );
   218→  }

   220→  void _toggleSelection(String id) {
   221→    setState(() {
   222→      if (_selectedIds.contains(id)) {
   223→        _selectedIds.remove(id);
   224→      } else {
   225→        _selectedIds.add(id);
   226→      }
   227→      if (_selectedIds.isEmpty) _isSelectionMode = false;
   228→    });
   229→  }
   230→
   231→  void _selectAll() {
   232→    setState(() {
   233→      if (_selectedIds.length == _filteredInvoices.length) {
   234→        _selectedIds.clear();
   235→      } else {
   236→        _selectedIds = _filteredInvoices.map((inv) => inv.id).toSet();
   237→      }
   238→      if (_selectedIds.isEmpty) _isSelectionMode = false;
   239→    });
   240→  }
   241→
   242→  Future<void> _deleteSelected() async {
   243→    if (_selectedIds.isEmpty) return;
   244→    final confirm = await showDialog<bool>(
   245→      context: context,
   246→      builder: (context) => AlertDialog(
   247→        shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(Constants.cardRadius)),
   248→        title: const Text('Confirmer la suppression'),
   249→        content: Text('Voulez-vous vraiment supprimer ${_selectedIds.length} facture(s) ?'),
   250→        actions: [
   251→          TextButton(onPressed: () => Navigator.pop(context, false), child: const Text('Annuler')),
   252→          TextButton(
   253→            onPressed: () => Navigator.pop(context, true),
   254→            child: const Text('Supprimer', style: TextStyle(color: Colors.red)),
   255→          ),
   256→        ],
   257→      ),
   258→    );
   259→    if (confirm != true) return;
   260→
   261→    final deletedCount = _selectedIds.length;
   262→    _invoices.removeWhere((inv) => _selectedIds.contains(inv.id));
   263→    await _storageService.saveInvoices(_invoices);
   264→    setState(() {
   265→      _filterInvoices();
   266→      _isSelectionMode = false;
   267→      _selectedIds.clear();
   268→      _displayedCount = 50;
   269→    });
   270→    if (!mounted) return;
   271→    ScaffoldMessenger.of(context).showSnackBar(
   272→      SnackBar(content: Text('$deletedCount facture(s) supprimée(s)'), backgroundColor: Colors.green,
   273→      behavior: SnackBarBehavior.floating),
   274→    );
   275→  }

   277→  /// Exporte en CSV la sélection actuellement filtrée (période, statut, recherche...).
   278→  Future<void> _exportFilteredCsv() async {
   279→    if (_filteredInvoices.isEmpty) return;
   280→    final backupService = BackupService();
   281→    final csv = backupService.invoicesToCsv(_filteredInvoices);
   282→    final filename =
   283→        'factures_${DateFormat('yyyyMMdd_HHmm').format(DateTime.now())}.csv';
   284→
   285→    ScaffoldMessenger.of(context).showSnackBar(
   286→      SnackBar(
   287→        content: Text('Export CSV de ${_filteredInvoices.length} facture(s)...'),
   288→        backgroundColor: const Color(0xFF2563EB),
   289→        behavior: SnackBarBehavior.floating,
   290→        duration: const Duration(seconds: 2),
   291→      ),
   292→    );
   293→    try {
   294→      await backupService.shareCsv(csv, filename);
   295→    } catch (e) {
   296→      if (!mounted) return;
   297→      ScaffoldMessenger.of(context).showSnackBar(
   298→        SnackBar(
   299→          content: Text('Erreur d\'export : $e'),
   300→          backgroundColor: Colors.red,
   301→          behavior: SnackBarBehavior.floating,
   302→        ),
   303→      );
   304→    }
   305→  }
   306→
   307→  Future<void> _exportSelected() async {
   308→    if (_selectedIds.isEmpty) return;
   309→    final selectedInvoices = _invoices.where((inv) => _selectedIds.contains(inv.id)).toList();
   310→
   311→    ScaffoldMessenger.of(context).showSnackBar(
   312→      SnackBar(
   313→        content: Text('Export de ${selectedInvoices.length} facture(s)...'),
   314→        backgroundColor: const Color(0xFF2563EB),
   315→        behavior: SnackBarBehavior.floating,
   316→        duration: const Duration(seconds: 2),
   317→      ),
   318→    );
   319→
   320→    // Un seul partage groupé (liste de XFile) au lieu d'ouvrir le
   321→    // partage natif une fois par facture.
   322→    try {
   323→      await _pdfService.shareInvoicesAsPdf(selectedInvoices);
   324→    } catch (e) {
   325→      if (!mounted) return;
   326→      ScaffoldMessenger.of(context).showSnackBar(
   327→        SnackBar(
   328→          content: Text('Erreur d\'export : $e'),
   329→          backgroundColor: Colors.red,
   330→          behavior: SnackBarBehavior.floating,
   331→        ),
   332→      );
   333→    }
   334→  }

   336→  void _exitSelectionMode() {
   337→    setState(() {
   338→      _isSelectionMode = false;
   339→      _selectedIds.clear();
   340→    });
   341→  }

   343→  void _clearStatusFilters() {
   344→    setState(() {
   345→      _filterEnCours = false;
   346→      _filterEnLivraison = false;
   347→      _filterLivree = false;
   348→      _filterArchivee = false;
   349→      _filterInvoices();

   351→  }

   353→  Future<void> _pickDateRange() async {
   354→    final now = DateTime.now();
   355→    final picked = await showDateRangePicker(
   356→      context: context,
   357→      firstDate: DateTime(now.year - 10),
   358→      lastDate: now,
   359→      initialDateRange: _filterDateRange,
   360→      helpText: 'Sélectionner une période',
   361→      saveText: 'Appliquer',
   362→      cancelText: 'Annuler',
   363→    );
   364→    if (picked == null || !mounted) return;
   365→    setState(() {
   366→      _filterDateRange = picked;
   367→      _filterInvoices();
   368→      _displayedCount = 50;
   369→    });
   370→  }
   371→
   372→  void _clearDateFilter() {
   373→    setState(() {
   374→      _filterDateRange = null;
   375→      _filterInvoices();
   376→      _displayedCount = 50;
   377→    });
   378→  }

   381→  double get _totalAmount => _filteredInvoices.fold(0.0, (sum, inv) => sum + inv.total);

   386→      InvoiceStatus.livree: _filteredInvoices.where((inv) => inv.status == InvoiceStatus.livree).length,
   387→      InvoiceStatus.archivee: _filteredInvoices.where((inv) => inv.status == InvoiceStatus.archivee).length,
   388→    };
   389→  }

   674→  Widget _buildSwipeableInvoiceCard(Invoice invoice, bool isSelected, int index) {
   675→    return Dismissible(
   676→      key: Key('swipe_${invoice.id}'),
   677→      confirmDismiss: (direction) async {
   678→        if (direction == DismissDirection.startToEnd) {
   679→          // Gauche → Droite : statut avance
   680→          InvoiceStatus newStatus;
   681→          switch (invoice.status) {
   682→            case InvoiceStatus.enCours:
   683→              newStatus = InvoiceStatus.enLivraison;
   684→              break;
   685→            case InvoiceStatus.enLivraison:
   686→              newStatus = InvoiceStatus.livree;
   687→              break;
   688→            case InvoiceStatus.livree:
   689→              newStatus = InvoiceStatus.archivee;
   690→              break;
   691→            case InvoiceStatus.archivee:
   692→              return false; // Ne rien faire si archivée
   693→          }
   694→          await _updateInvoiceStatus(invoice, newStatus);
   695→          return false; // Ne pas supprimer la carte
   696→        } else {
   697→          // Droite → Gauche : statut recule
   698→          InvoiceStatus newStatus;
   699→          switch (invoice.status) {
   700→            case InvoiceStatus.livree:
   701→              newStatus = InvoiceStatus.enLivraison;
   702→              break;
   703→            case InvoiceStatus.enLivraison:
   704→              newStatus = InvoiceStatus.enCours;
   705→              break;
   706→            case InvoiceStatus.enCours:
   707→              return false; // Déjà au minimum
   708→            case InvoiceStatus.archivee:
   709→              newStatus = InvoiceStatus.livree;
   710→              break;
   711→          }
   712→          await _updateInvoiceStatus(invoice, newStatus);
   713→          return false; // Ne pas supprimer la carte
   714→        }
   715→      },
   716→      background: Container(
   717→        alignment: Alignment.centerLeft,
   718→        padding: const EdgeInsets.only(left: 20),
   719→        decoration: BoxDecoration(
   720→          borderRadius: BorderRadius.circular(Constants.cardRadius),
   721→          gradient: const LinearGradient(
   722→            colors: [Color(0xFF16A34A), Color(0xFF15803D)],
   723→            begin: Alignment.centerLeft,
   724→            end: Alignment.centerRight,
   725→          ),
   726→        ),
   727→        child: const Row(

   764→  Future<void> _updateInvoiceStatus(Invoice invoice, InvoiceStatus newStatus) async {
   765→    final updatedInvoice = invoice.copyWith(status: newStatus);
   766→    final all = await _storageService.loadInvoices();
   767→    final idx = all.indexWhere((inv) => inv.id == invoice.id);
   768→    if (idx != -1) {
   769→      all[idx] = updatedInvoice;
   770→      await _storageService.saveInvoices(all);
   771→    }
   772→    if (!mounted) return;
   773→    setState(() {
   774→      final listIdx = _invoices.indexWhere((inv) => inv.id == invoice.id);
   775→      if (listIdx != -1) {
   776→        _invoices[listIdx] = updatedInvoice;
   777→      }
   778→      _filterInvoices();
   779→    });
   780→    widget.onInvoiceUpdated();
   781→    if (!mounted) return;
   782→    ScaffoldMessenger.of(context).showSnackBar(
   783→      SnackBar(
   784→        content: Text('${invoice.clientName} → ${newStatus.label}'),
   785→        backgroundColor: Color(newStatus.color),
   786→        behavior: SnackBarBehavior.floating,
   787→        duration: const Duration(seconds: 2),
   788→      ),
   789→    );
   790→  }
