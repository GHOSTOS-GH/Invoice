    13→class NewInvoiceScreen extends StatefulWidget {
    14→  final VoidCallback onInvoiceCreated;
    15→  final Invoice? initialInvoice;
    16→  const NewInvoiceScreen({super.key, required this.onInvoiceCreated, this.initialInvoice});
    17→
    18→  @override
    19→  State<NewInvoiceScreen> createState() => _NewInvoiceScreenState();
    20→}
    21→
    22→class _NewInvoiceScreenState extends State<NewInvoiceScreen> with WidgetsBindingObserver {
    23→  final _clientNameController = TextEditingController();
    24→  final _quantityController = TextEditingController();
    25→  final _priceController = TextEditingController();
    26→  final _storageService = StorageService();
    27→  final _historyService = HistoryService();
    28→  final List<InvoiceItem> _items = [];
    29→  bool _isAddingItem = false;
    30→  Product? _selectedProduct;
    31→  bool _isLoading = false;
    32→  List<String> _clientSuggestions = [];
    33→  bool _showClientSuggestions = false;
    34→  final FocusNode _clientFocusNode = FocusNode();
    35→
    36→  // Dernier prix par produit (clé : nom du produit en minuscules) pour le client saisi
    37→  Map<String, double> _clientPrices = {};
    38→  String _lastPriceClient = '';
    39→  int _priceLoadToken = 0;
    40→
    41→  @override
    42→  void initState() {
    43→    super.initState();
    44→    WidgetsBinding.instance.addObserver(this);
    45→    _clientNameController.addListener(_onClientNameChanged);
    46→    _loadInitialData();
    47→    _clientFocusNode.addListener(() {
    48→      if (!_clientFocusNode.hasFocus) {
    49→        setState(() => _showClientSuggestions = false);
    50→      }
    51→    });
    52→  }
    53→
    54→  Future<void> _loadInitialData() async {
    55→    if (widget.initialInvoice != null) {
    56→      _clientNameController.text = widget.initialInvoice!.clientName;
    57→      _items.addAll(widget.initialInvoice!.items.map((item) => item.copyWith()));
    58→    } else {
    59→      final draft = await _storageService.loadDraft();
    60→      if (!mounted) return;
    61→      if (draft != null) {
    62→        _clientNameController.text = draft.clientName;
    63→        _items.addAll(draft.items);
    64→      }
    65→    }
    66→    setState(() {});
    67→  }
    68→
    69→  @override
    70→  void didChangeAppLifecycleState(AppLifecycleState state) {
    71→    if (state == AppLifecycleState.paused || state == AppLifecycleState.inactive) {
    72→      _saveDraft();
    73→    }
    74→  }
    75→
    76→  @override
    77→  void dispose() {
    78→    WidgetsBinding.instance.removeObserver(this);
    79→    _clientNameController.removeListener(_onClientNameChanged);
    80→    _clientNameController.dispose();
    81→    _quantityController.dispose();
    82→    _priceController.dispose();
    83→    _clientFocusNode.dispose();
    84→    _saveDraft();
    85→    super.dispose();
    86→  }
    87→
    88→  Future<void> _saveDraft() async {
    89→    if (_items.isEmpty && _clientNameController.text.trim().isEmpty) {
    90→      await _storageService.clearDraft();
    91→      return;
    92→    }
    93→    final draft = Invoice(
    94→      id: 'draft',
    95→      clientName: _clientNameController.text.trim().isEmpty
    96→          ? 'Client inconnu'
    97→          : _clientNameController.text.trim(),
    98→      createdAt: DateTime.now(),
    99→      items: List.from(_items),
   100→    );
   101→    await _storageService.saveDraft(draft);
   102→  }
   103→
   104→  void _onClientNameChanged() async {
   105→    final clientText = _clientNameController.text;
   106→    if (clientText.isEmpty) {
   107→      setState(() {
   108→        _showClientSuggestions = false;
   109→        _clientPrices = {};
   110→        _lastPriceClient = '';
   111→      });
   112→      return;
   113→    }
   114→    final suggestions = await _historyService.searchClientNames(clientText);
   115→    if (!mounted) return;
   116→
   117→    // Charge (une seule fois par nom de client) la table des derniers prix
   118→    if (clientText.trim().toLowerCase() != _lastPriceClient) {
   119→      final token = ++_priceLoadToken;
   120→      final prices = await _storageService.getClientPriceMap(clientText);
   121→      if (!mounted || token != _priceLoadToken) return;
   122→      _clientPrices = prices;
   123→      _lastPriceClient = clientText.trim().toLowerCase();
   124→    }
   125→
   126→    setState(() {
   127→      _clientSuggestions = suggestions;
   128→      _showClientSuggestions = suggestions.isNotEmpty && _clientFocusNode.hasFocus;
   129→    });
   130→  }
   131→
   132→  void _addItem() {
   133→    final quantity = int.tryParse(_quantityController.text.trim());
   134→    if (quantity == null || quantity <= 0) {
   135→      ScaffoldMessenger.of(context).showSnackBar(
   136→        const SnackBar(
   137→          content: Text('Veuillez entrer une quantité valide'),
   138→          backgroundColor: Colors.red,
   139→          behavior: SnackBarBehavior.floating,
   140→        ),
   141→      );
   142→      return;
   143→    }
   144→
   145→    double? price = double.tryParse(_priceController.text.trim());
   146→    if (price == null || price <= 0) {
   147→      price = 1.0;
   148→      _priceController.text = '1';
   149→    }
   150→
   151→    if (_selectedProduct == null) {
   152→      ScaffoldMessenger.of(context).showSnackBar(
   153→        const SnackBar(
   154→          content: Text('Veuillez sélectionner un produit'),
   155→          backgroundColor: Colors.orange,
   156→        ),
   157→      );
   158→      return;
   159→    }
   160→
   161→    final productName = _selectedProduct!.name;
   162→    final item = InvoiceItem(
   163→      name: productName,
   164→      quantity: quantity,
   165→      unitPrice: price,
   166→    );
   167→
   168→    setState(() {
   169→      _items.add(item);
   170→      _quantityController.clear();
   171→      _priceController.clear();
   172→      _selectedProduct = null;
   173→      _isAddingItem = false;
   174→    });
   175→
   176→    _historyService.saveProductName(productName);
   177→    _saveDraft();
   178→  }
   179→
   180→  void _editItem(int index) {
   181→    showDialog(
   182→      context: context,
   183→      builder: (_) => _ItemEditDialog(
   184→        item: _items[index],
   185→        onSave: (updated) {
   186→          setState(() => _items[index] = updated);
   187→          _saveDraft();
   188→        },
   189→      ),
   190→    );
   191→  }
   192→
   193→  void _deleteItem(int index) {
   194→    setState(() => _items.removeAt(index));
   195→    _saveDraft();
   196→  }
   197→
   198→  double _calculateTotal() => _items.fold(0.0, (sum, item) => sum + item.subtotal);
   199→
   200→  Future<void> _finalizeInvoice() async {
   201→    if (_items.isEmpty) {
   202→      ScaffoldMessenger.of(context).showSnackBar(
   203→        const SnackBar(
   204→          content: Text('Ajoutez au moins un article avant de finaliser'),
   205→          backgroundColor: Colors.orange,
   206→          behavior: SnackBarBehavior.floating,
   207→        ),
   208→      );
   209→      return;
   210→    }
   211→
   212→    setState(() => _isLoading = true);
   213→
   214→    final clientName = _clientNameController.text.trim().isEmpty
   215→        ? 'Client inconnu'
   216→        : _clientNameController.text.trim();
   217→
   218→    final invoice = Invoice(
   219→      id: DateTime.now().millisecondsSinceEpoch.toString(),
   220→      clientName: clientName,
   221→      createdAt: DateTime.now(),
   222→      items: List.from(_items),
   223→    );
   224→
   225→    try {
   226→      await _storageService.addInvoice(invoice);
   227→      if (clientName != 'Client inconnu') {
   228→        await _historyService.saveClientName(clientName);
   229→      }
   230→      await _storageService.clearDraft();
   231→
   232→      if (!mounted) return;
   233→
   234→      setState(() {
   235→        _clientNameController.clear();
   236→        _items.clear();
   237→        _isAddingItem = false;
   238→        _isLoading = false;
   239→      });
   240→
   241→      ScaffoldMessenger.of(context).showSnackBar(
   242→        const SnackBar(
   243→          content: Text('Facture enregistrée avec succès !'),
   244→          backgroundColor: Colors.green,
   245→          behavior: SnackBarBehavior.floating,
   246→        ),
   247→      );
   248→      widget.onInvoiceCreated();
   249→    } catch (e) {
   250→      if (!mounted) return;
   251→      setState(() => _isLoading = false);
   252→      ScaffoldMessenger.of(context).showSnackBar(
   253→        SnackBar(
   254→          content: Text('Erreur lors de la sauvegarde: $e'),
   255→          backgroundColor: Colors.red,
   256→          behavior: SnackBarBehavior.floating,
   257→        ),
   258→      );
   259→    }
   260→  }

   273→  void _onProductSelected(Product product) {
   274→    setState(() {
   275→      final isSameProduct = _selectedProduct?.id == product.id;
   276→      _selectedProduct = product;
   277→      if (!isSameProduct) {
   278→        // Pré-remplit le prix avec le dernier prix vendu à ce client
   279→        final lastPrice = _clientPrices[product.name.trim().toLowerCase()];
   280→        if (lastPrice != null) {
   281→          _priceController.text = _priceInputValue(lastPrice);
   282→        } else {
   283→          _priceController.clear();
   284→        }
   285→      }
   286→    });
   287→  }
   288→
   289→  String _priceInputValue(double value) {
   290→    return value % 1 == 0 ? value.toInt().toString() : value.toString();
   291→  }

   694→class _ItemEditDialog extends StatefulWidget {
   695→  final InvoiceItem item;
   696→  final Function(InvoiceItem) onSave;
   697→
   698→  const _ItemEditDialog({required this.item, required this.onSave});
   699→
   700→  @override
   701→  State<_ItemEditDialog> createState() => _ItemEditDialogState();
   702→}
   703→
   704→class _ItemEditDialogState extends State<_ItemEditDialog> {
   705→  late TextEditingController _nameController;
   706→  late TextEditingController _quantityController;
   707→  late TextEditingController _priceController;
   708→
   709→  @override
   710→  void initState() {
   711→    super.initState();
   712→    _nameController = TextEditingController(text: widget.item.name);
   713→    _quantityController = TextEditingController(text: widget.item.quantity.toString());
   714→    _priceController = TextEditingController(text: widget.item.unitPrice.toString());
   715→  }
   716→
   717→  @override
   718→  void dispose() {
   719→    _nameController.dispose();
   720→    _quantityController.dispose();
   721→    _priceController.dispose();
   722→    super.dispose();
   723→  }
   724→
   725→  void _save() {
   726→    final quantity = int.tryParse(_quantityController.text);
   727→    final price = double.tryParse(_priceController.text);
   728→    if (quantity == null || quantity <= 0 || price == null || price <= 0) {
   729→      ScaffoldMessenger.of(context).showSnackBar(
   730→        const SnackBar(content: Text('Valeurs invalides'), backgroundColor: Colors.red),
   731→      );
   732→      return;
   733→    }
   734→    final updated = widget.item.copyWith(
   735→      name: _nameController.text.trim().isEmpty ? 'Sans nom' : _nameController.text.trim(),
   736→      quantity: quantity,
   737→      unitPrice: price,
   738→    );
   739→    widget.onSave(updated);
   740→    Navigator.pop(context);
   741→  }
   742→
   743→  @override
   744→  Widget build(BuildContext context) {
   745→    return AlertDialog(
   746→      title: const Text('Modifier l\'article'),
   747→      content: Column(
   748→        mainAxisSize: MainAxisSize.min,
   749→        children: [
   750→          TextField(
   751→            controller: _nameController,
   752→            decoration: InputDecoration(
   753→              labelText: 'Nom',
   754→              border: OutlineInputBorder(borderRadius: BorderRadius.circular(14)),
   755→            ),
   756→          ),
   757→          const SizedBox(height: 12),
   758→          Row(
   759→            children: [
   760→              Expanded(
   761→                child: TextField(
   762→                  controller: _quantityController,
   763→                  keyboardType: TextInputType.number,
   764→                  inputFormatters: [FilteringTextInputFormatter.digitsOnly],
   765→                  decoration: InputDecoration(
   766→                    labelText: 'Quantité',
   767→                    border: OutlineInputBorder(borderRadius: BorderRadius.circular(14)),
   768→                  ),
   769→                ),
   770→              ),
   771→              const SizedBox(width: 12),
   772→              Expanded(
   773→                child: TextField(
   774→                  controller: _priceController,
   775→                  keyboardType: const TextInputType.numberWithOptions(decimal: true),
   776→                  inputFormatters: [FilteringTextInputFormatter.allow(RegExp(r'^\d*\.?\d*'))],
   777→                  decoration: InputDecoration(
   778→                    labelText: 'Prix unitaire',
   779→                    border: OutlineInputBorder(borderRadius: BorderRadius.circular(14)),
   780→                  ),
   781→                ),
   782→              ),
   783→            ],
   784→          ),
   785→        ],
   786→      ),
   787→      actions: [
   788→        TextButton(onPressed: () => Navigator.pop(context), child: const Text('Annuler')),
   789→        ElevatedButton(onPressed: _save, child: const Text('Enregistrer')),
   790→      ],
   791→    );
   792→  }
